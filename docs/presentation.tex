\begin{frame}{Architecture Overview}
  \begin{columns}[T]

    \column{0.48\textwidth}
    \textbf{\color{accentcol}The Challenge}\\[0.3em]
    {\small We use dozens of apps, devices, and services every day —
    but nothing connects them into one smart system that works for \emph{us}.}

    \vspace{1em}
    \textbf{\color{accentcol}The System}\\[0.3em]
    {\small CollectiveOS is a personal AI hub. It sits at the centre of your
    digital life, understands what you need, takes action, and remembers
    your preferences across every session.}

    \vspace{1em}
    \textbf{\color{accentcol}Security \& Control}\\[0.3em]
    {\small Every sensitive action requires your explicit approval.
    Your data stays local. Credentials are encrypted. Nothing happens
    behind your back.}

    \column{0.48\textwidth}
    \textbf{\color{accentcol}Key Design Decisions}
    \begin{itemize}\small
      \item Safety first — write actions need your sign-off
      \item Memory-first — learns and remembers across sessions
      \item Modular — add any new service without changing the core
      \item Single user, fully private
    \end{itemize}

    \vspace{0.8em}
    \textbf{\color{accentcol}Tech Snapshot}
    \begin{itemize}\small
      \item \textbf{AI:} Google Gemini (language + vision)
      \item \textbf{Memory:} PostgreSQL + vector search
      \item \textbf{API:} FastAPI (Python)
      \item \textbf{Automation:} LangGraph agent loop
    \end{itemize}

    \vspace{0.8em}
    \textbf{\color{accentcol}Integrations Planned}
    {\small Calendar · Email · Smart Home · Web · Files · Vehicle}

  \end{columns}
\end{frame}

% ─── 7. CURRENT LIMITATIONS ──────────────────────────────────────────────────
\begin{frame}{Where We Are Honest About Limitations}
  \begin{columns}[T]
    \column{0.48\textwidth}
    {\color{accentcol}\bfseries Current constraints:}
    \begin{itemize}\small
      \item \textbf{Service connectors:} Calendar and email are next — not yet wired
      \item \textbf{Single user:} Built for one person; no multi-account support yet
      \item \textbf{Desktop only:} No mobile app; phone integration is planned
      \item \textbf{Local setup:} Requires a Mac with the server running
      \item \textbf{Web navigation speed:} Visual-only browsing is slower than
            dedicated browser APIs
    \end{itemize}

    \column{0.48\textwidth}
    {\color{accentcol}\bfseries Why this is fine for now:}
    \begin{itemize}\small
      \item Each limitation is a \emph{known, planned milestone} — not a
            fundamental blocker
      \item The core loop (understand → decide → act → remember) is solid
            and tested
      \item Safety constraints are intentional: read-only comes before write;
            approval gates stay forever
      \item Every new service is one connector file — the architecture
            scales cleanly
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Technical Walkthrough — One Complete Journey}
  \vspace{0.3em}
  {\small\color{subtext}\itshape
    ``In Safari, search for the current weather'' — traced from input to result}
  \vspace{0.5em}

  \begin{enumerate}\small
    \item \textbf{Input} \quad User types the request in the chat window
    \item \textbf{Processing} \quad AI router classifies intent as a
          \emph{computer-use} task; semantic memory is checked for past
          navigation patterns
    \item \textbf{Intelligence / Logic} \quad Task agent plans steps:
          (a) open Safari, (b) navigate to a weather site; generates
          an approval request before acting
    \item \textbf{Tools} \quad On approval, the computer agent uses
          vision + accessibility tree to click, type, and navigate —
          verifying each step succeeded
    \item \textbf{Output} \quad Result displayed in chat; screen
          screenshots streamed live to the Computer panel
    \item \textbf{Logs / Observability} \quad Session saved to PostgreSQL,
          navigation pattern stored in vector memory, token cost logged
  \end{enumerate}

  \vspace{0.3em}
  {\scriptsize\color{subtext}
    Every step is traceable. If anything goes wrong, the agent retries
    up to twice, then reports the failure honestly.}
\end{frame}
